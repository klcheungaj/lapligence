function(strict_c target)
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    # -Wshadow mirrors MSVC /W4 C4456/C4457/C4459, which the Windows lanes
    # reject through /WX.
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wshadow -Werror -pedantic-errors)
  endif()
endfunction()

# against the selected compact descriptor, as generated models do; the compact
# units allocate through the tracked hooks of tracked_value.c/storage_runtime.c.

if(LLG_STORAGE_TEST_VALUE_BACKEND STREQUAL "compact")
  set(LLG_TEST_VALUE_COMPACT ON)
  if(LLG_STORAGE_TEST_COMPACT)
    message(FATAL_ERROR "LLG_STORAGE_TEST_COMPACT links legacy and compact together; configure it with the legacy backend")
  endif()
  set(selected_value_libraries)
  if(NOT MSVC)
    list(APPEND selected_value_libraries m)
  endif()
  if(LLG_STORAGE_TEST_COMPACT_KERNELS STREQUAL "gmp")
    find_path(selected_gmp_include gmp.h PATHS "${LLG_GMP_ROOT}/include" NO_DEFAULT_PATH)
    find_library(selected_gmp_library gmp PATHS "${LLG_GMP_ROOT}/lib" NO_DEFAULT_PATH)
    if(NOT LLG_GMP_ROOT OR NOT selected_gmp_include OR NOT selected_gmp_library)
      message(FATAL_ERROR "compact gmp kernels require LLG_GMP_ROOT with include/gmp.h and lib/libgmp")
    endif()
    list(APPEND selected_value_libraries "${selected_gmp_library}")
    add_compile_definitions(LLG_SV4_USE_GMP=1 LLG_SV4_GMP_KERNELS=1)
  elseif(LLG_STORAGE_TEST_COMPACT_KERNELS STREQUAL "portable")
    add_compile_definitions(LLG_SV4_USE_GMP=1 LLG_SV4_GMP_KERNELS=0)
  else()
    message(FATAL_ERROR "LLG_STORAGE_TEST_COMPACT_KERNELS must be portable or gmp")
  endif()
  set(selected_value_units storage logic arithmetic shifts_reductions comparison_membership
    selections references assembly consumer_bridge kernels net_adapters real_time format_index)
  list(TRANSFORM selected_value_units PREPEND "${LLG_RT}/value_gmp/")
  list(TRANSFORM selected_value_units APPEND ".c")
  add_library(selected_value_units OBJECT ${selected_value_units})
  target_include_directories(selected_value_units PRIVATE "${LLG_RT}")
  if(selected_gmp_include)
    target_include_directories(selected_value_units PRIVATE "${selected_gmp_include}")
  endif()
  target_compile_definitions(selected_value_units PRIVATE malloc=llg_value_test_malloc
    calloc=llg_value_test_calloc realloc=llg_value_test_realloc free=llg_value_test_free)
elseif(NOT LLG_STORAGE_TEST_VALUE_BACKEND STREQUAL "legacy")
  message(FATAL_ERROR "LLG_STORAGE_TEST_VALUE_BACKEND must be legacy or compact")
endif()
