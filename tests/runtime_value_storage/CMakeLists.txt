if(LLG_STORAGE_TEST_SANITIZERS)
  set(LLG_TEST_HAS_SANITIZERS true)
  if(MSVC OR NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "This sanitizer option requires a GCC/Clang-style compiler driver")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

function(strict_c target)
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic-errors)
  endif()
endfunction()

add_executable(sv4_storage_probe storage_probe.c)
target_include_directories(sv4_storage_probe PRIVATE "${LLG_RT}")
strict_c(sv4_storage_probe)
add_test(NAME storage_lifecycle COMMAND sv4_storage_probe)
foreach(mode IN ITEMS limit uint32-max oom oom-copy)
  if(mode MATCHES "oom")
    set(expected "packed storage allocation failed")
  else()
    set(expected "packed storage width")
  endif()
  add_test(NAME storage_reject_${mode}
    COMMAND "${CMAKE_COMMAND}"
      "-DPROBE=$<TARGET_FILE:sv4_storage_probe>" "-DMODE=${mode}"
      "-DEXPECTED=${expected}" -P "${CMAKE_CURRENT_SOURCE_DIR}/expect_failure.cmake")
endforeach()

  set(LLG_TEST_HAS_WAVEFORMS true)
  find_package(Threads REQUIRED)

    "${LLG_RT}/gtkwave/fstapi.c"

  add_executable(sv4_waveform_probe waveform_snapshot_probe.c storage_runtime.c)

  target_link_libraries(sv4_waveform_probe PRIVATE storage_probe_fst)

option(LLG_STORAGE_TEST_SCHEDULER "Test retained scheduler ownership (requires vendored libaco on a supported host)" ON)
if(LLG_STORAGE_TEST_SCHEDULER)
  if(UNIX AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|i.86")
    set(LLG_TEST_HAS_SCHEDULER true)
    enable_language(ASM)
    add_library(owner_probe_aco STATIC "${LLG_ROOT}/vendor/libaco/aco.c" "${LLG_ROOT}/vendor/libaco/acosw.S")
    target_include_directories(owner_probe_aco PUBLIC "${LLG_ROOT}/vendor/libaco")
    add_library(owner_probe_container STATIC "${LLG_RT}/llg_container.c")
    target_link_libraries(owner_probe_container PUBLIC tracked_value)
    strict_c(owner_probe_container)
    add_executable(vpi_ownership_probe vpi_ownership_probe.c)
    target_link_libraries(vpi_ownership_probe PRIVATE owner_probe_aco owner_probe_container ${CMAKE_DL_LIBS})
    strict_c(vpi_ownership_probe)
    add_test(NAME vpi_ownership COMMAND vpi_ownership_probe)
    set_tests_properties(vpi_ownership PROPERTIES TIMEOUT 60)
    add_executable(scheduler_ownership_probe scheduler_ownership_probe.c)
    target_link_libraries(scheduler_ownership_probe PRIVATE owner_probe_aco owner_probe_container)
    strict_c(scheduler_ownership_probe)
    add_test(NAME scheduler_ownership COMMAND scheduler_ownership_probe)
    set_tests_properties(scheduler_ownership PROPERTIES TIMEOUT 60)
    add_executable(generated_scopes_probe generated_scopes_probe.c)
    target_link_libraries(generated_scopes_probe PRIVATE owner_probe_aco owner_probe_container)
    strict_c(generated_scopes_probe)
    add_test(NAME generated_scope_patterns COMMAND generated_scopes_probe)
    set_tests_properties(generated_scope_patterns PROPERTIES TIMEOUT 60)
    # libaco does not advertise sanitizer fiber-switch integration. Keep real
    # stack-switch tests separate from the sanitizer-safe ownership unit probe.
    if(NOT LLG_STORAGE_TEST_SANITIZERS)
      set(LLG_TEST_HAS_COROUTINES true)
      add_executable(coroutine_ownership_probe coroutine_ownership_probe.c)
      target_link_libraries(coroutine_ownership_probe PRIVATE owner_probe_aco owner_probe_container)
      strict_c(coroutine_ownership_probe)
      add_test(NAME coroutine_ownership COMMAND coroutine_ownership_probe)
      set_tests_properties(coroutine_ownership PROPERTIES TIMEOUT 60)
      add_executable(generated_coroutine_probe generated_coroutine_probe.c)
      target_link_libraries(generated_coroutine_probe PRIVATE owner_probe_aco owner_probe_container)
      strict_c(generated_coroutine_probe)
      add_test(NAME generated_coroutine_patterns COMMAND generated_coroutine_probe)
      set_tests_properties(generated_coroutine_patterns PROPERTIES TIMEOUT 60)
    endif()
  else()
    message(STATUS "Scheduler probe skipped: libaco native test support is limited to Unix x86 hosts")
  endif()
endif()
