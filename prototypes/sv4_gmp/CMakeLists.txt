cmake_minimum_required(VERSION 3.16)
project(llg_sv4_prototype LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(LLG_SV4_USE_GMP "Select GMP for the prototype client (not llg production)" OFF)
option(LLG_SV4_BUILD_DIFFERENTIAL "Build both independent backends and comparison tests" ${LLG_SV4_USE_GMP})
option(LLG_SV4_SANITIZERS "Enable ASan/UBSan on GCC/Clang" OFF)
set(GMP_ROOT "" CACHE PATH "Prefix of a target-compatible GMP 6.3.0 installation")
include(CTest)

if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror -Wvla)
endif()
if(LLG_SV4_SANITIZERS)
  if(MSVC)
    message(FATAL_ERROR "This sanitizer recipe requires GCC/Clang; native MSVC is not qualified")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

function(add_golden)
  add_library(sv4_golden STATIC golden/llg_value.c)
  target_include_directories(sv4_golden PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/golden")
  if(NOT MSVC)
    target_link_libraries(sv4_golden PUBLIC m)
  endif()
endfunction()
if(NOT LLG_SV4_USE_GMP OR LLG_SV4_BUILD_DIFFERENTIAL)
  add_golden()
endif()

if(LLG_SV4_USE_GMP OR LLG_SV4_BUILD_DIFFERENTIAL)
  unset(GMP_INCLUDE_DIR CACHE)
  unset(GMP_LIBRARY CACHE)
  if(GMP_ROOT)
    find_path(GMP_INCLUDE_DIR gmp.h PATHS "${GMP_ROOT}/include" NO_DEFAULT_PATH)
    find_library(GMP_LIBRARY NAMES gmp libgmp PATHS "${GMP_ROOT}/lib" "${GMP_ROOT}/lib64" NO_DEFAULT_PATH)
  else()
    find_path(GMP_INCLUDE_DIR gmp.h)
    find_library(GMP_LIBRARY NAMES gmp libgmp)
  endif()
  if(NOT GMP_INCLUDE_DIR OR NOT GMP_LIBRARY)
    message(FATAL_ERROR "GMP was not found. Set GMP_ROOT to a matching installation; explicit roots never fall back to the system.")
  endif()
  include(CheckCSourceCompiles)
  set(CMAKE_REQUIRED_INCLUDES "${GMP_INCLUDE_DIR}")
  set(CMAKE_REQUIRED_LIBRARIES "${GMP_LIBRARY}")
  unset(LLG_GMP_API_OK CACHE)
  check_c_source_compiles("
    #include <gmp.h>
    #if __GNU_MP_VERSION != 6 || __GNU_MP_VERSION_MINOR != 3 || __GNU_MP_VERSION_PATCHLEVEL != 0
    #error This experiment is pinned to the attached GMP 6.3.0
    #endif
    _Static_assert(GMP_NAIL_BITS == 0, \"nails unsupported\");
    _Static_assert(GMP_NUMB_BITS == 32 || GMP_NUMB_BITS == 64, \"limbs unsupported\");
    int main(void) {
      mp_limb_t a[1]={1},b[1]={2},r[2]={0};
      mpn_add_n(r,a,b,1); mpn_sub_n(r,a,b,1); mpn_mul_n(r,a,b,1);
      mpn_sqr(r,a,1); mpn_neg(r,a,1); mpn_and_n(r,a,b,1);
      mpn_ior_n(r,a,b,1); mpn_xor_n(r,a,b,1); mpn_xnor_n(r,a,b,1);
      return 0;
    }" LLG_GMP_API_OK)
  if(NOT LLG_GMP_API_OK)
    message(FATAL_ERROR "Need a matching GMP 6.3.0 header/library and supported limb ABI. See CMakeError.log.")
  endif()
  if(NOT CMAKE_CROSSCOMPILING)
    include(CheckCSourceRuns)
    unset(LLG_GMP_RUNTIME_OK CACHE)
    check_c_source_runs("
      #include <gmp.h>
      #include <string.h>
      int main(void) {
        return strcmp(gmp_version, \"6.3.0\") != 0 || mp_bits_per_limb != GMP_NUMB_BITS;
      }" LLG_GMP_RUNTIME_OK)
    if(NOT LLG_GMP_RUNTIME_OK)
      message(FATAL_ERROR "Linked GMP version or limb size disagrees with the pinned header")
    endif()
  else()
    message(STATUS "Cross compiling: version/limb run probe must execute on the target")
  endif()
  add_library(GMP::GMP UNKNOWN IMPORTED)
  set_target_properties(GMP::GMP PROPERTIES IMPORTED_LOCATION "${GMP_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${GMP_INCLUDE_DIR}")
  set(GMP4_SOURCES src/storage.c src/conversion.c src/arithmetic.c src/logic.c src/wire.c)
  add_library(sv4_gmp STATIC ${GMP4_SOURCES})
  target_include_directories(sv4_gmp PUBLIC include)
  target_link_libraries(sv4_gmp PUBLIC GMP::GMP)
endif()

add_library(sv4_selected INTERFACE)
target_include_directories(sv4_selected INTERFACE include)
if(LLG_SV4_USE_GMP)
  target_compile_definitions(sv4_selected INTERFACE LLG_SV4_USE_GMP=1)
  target_link_libraries(sv4_selected INTERFACE sv4_gmp)
else()
  target_compile_definitions(sv4_selected INTERFACE LLG_SV4_USE_GMP=0)
  target_link_libraries(sv4_selected INTERFACE sv4_golden)
endif()
add_executable(sv4_sample tests/sample_model.c)

target_link_libraries(sv4_selected_benchmark PRIVATE sv4_selected)
set_target_properties(sv4_selected_benchmark PROPERTIES INTERPROCEDURAL_OPTIMIZATION FALSE)
if(BUILD_TESTING)
  find_package(Python3 3.8 COMPONENTS Interpreter REQUIRED)

  add_test(NAME golden_integrity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/verify_golden.py")
  add_test(NAME golden_verifier_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/golden_verifier.py")
  if(LLG_SV4_BUILD_DIFFERENTIAL)
    add_executable(sv4_differential tests/differential.c)
    target_link_libraries(sv4_differential PRIVATE sv4_gmp sv4_golden)
    add_test(NAME differential COMMAND sv4_differential)
  endif()
endif()

  target_include_directories(sv4_oracle PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(sv4_oracle PRIVATE sv4_gmp sv4_golden)
  add_test(NAME independent_math_oracle COMMAND sv4_oracle)
  add_executable(sv4_lifetime tests/lifetime.c)
  target_link_libraries(sv4_lifetime PRIVATE sv4_gmp)
  add_test(NAME lifetime_boundaries COMMAND sv4_lifetime)
  add_library(sv4_gmp_instrumented STATIC ${GMP4_SOURCES})
  target_include_directories(sv4_gmp_instrumented PUBLIC include PRIVATE tests)
  target_compile_definitions(sv4_gmp_instrumented PRIVATE GMP4_ALLOC_HEADER="allocation_hooks.h")

  add_executable(sv4_allocations tests/allocation_probe.c)
  target_link_libraries(sv4_allocations PRIVATE sv4_gmp_instrumented)
  add_test(NAME allocation_balance COMMAND sv4_allocations)
  add_executable(sv4_bad_args tests/bad_args.c)
  target_link_libraries(sv4_bad_args PRIVATE sv4_gmp)
  add_test(NAME expected_failures COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_failures.py"
    "$<TARGET_FILE:sv4_allocations>" "$<TARGET_FILE:sv4_bad_args>")
endif()
if(LLG_SV4_BUILD_DIFFERENTIAL)
  add_executable(sv4_benchmark tests/benchmark.c)
  target_link_libraries(sv4_benchmark PRIVATE sv4_gmp sv4_golden)
